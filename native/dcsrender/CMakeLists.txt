cmake_minimum_required(VERSION 3.20)
project(DcsRender LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
add_library(DcsRender SHARED src/scene_events.cpp)
target_compile_options(DcsRender PRIVATE /W4 /O2 /EHsc /permissive-)
target_compile_definitions(DcsRender PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(DcsRenderTests test/scene_events_test.cpp)
add_library(DcsRenderFakeBackend SHARED test/fake_backend.cpp)
add_library(DcsRenderFakeStereo SHARED test/fake_backend.cpp)
target_compile_definitions(DcsRenderFakeStereo PRIVATE TEST_STEREO)
set_target_properties(DcsRenderFakeBackend PROPERTIES OUTPUT_NAME dxgi RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/test-backend)
set_target_properties(DcsRenderFakeStereo PROPERTIES OUTPUT_NAME CheekyFoveatedDLSSRuntime RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/test-backend)
target_compile_options(DcsRenderTests PRIVATE /W4 /EHsc)
add_dependencies(DcsRenderTests DcsRender DcsRenderFakeBackend DcsRenderFakeStereo)
enable_testing()
add_test(NAME SceneEvents COMMAND DcsRenderTests $<TARGET_FILE:DcsRender> $<TARGET_FILE:DcsRenderFakeBackend>)
add_test(NAME StereoSceneEvents COMMAND DcsRenderTests $<TARGET_FILE:DcsRender> $<TARGET_FILE:DcsRenderFakeStereo> stereo)
